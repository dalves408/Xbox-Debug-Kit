cmake_minimum_required(VERSION 3.16)
project(xrdbg C)
# Builds only the demo/self-test. A game does not use this file: it includes cmake/xrdbg.cmake instead (see README.md).
include(cmake/xrdbg.cmake)
add_executable(xrdbg_demo WIN32 demo/xrdbg_demo.c)
xrdbg_enable(xrdbg_demo SAVE_STATES)
set_property(TARGET xrdbg_demo PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")

include(CTest)
if(BUILD_TESTING)
    add_executable(xrdbg_banner_test tests/banner_test.c)
    target_include_directories(xrdbg_banner_test PRIVATE include)
    target_link_libraries(xrdbg_banner_test PRIVATE dbghelp psapi shell32 user32 gdi32 advapi32)
    set_property(TARGET xrdbg_banner_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    add_test(NAME xrdbg_banner COMMAND xrdbg_banner_test)
endif()

if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME xrdbg_collector COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/collector_test.py")
    endif()
endif()

if(BUILD_TESTING)
    add_executable(xrdbg_state_test tests/state_test.c src/xrdbg_state.c)
    target_include_directories(xrdbg_state_test PRIVATE include)
    set_property(TARGET xrdbg_state_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    add_test(NAME xrdbg_state COMMAND xrdbg_state_test)
endif()

if(BUILD_TESTING)
    add_executable(xrdbg_demo_state_test tests/demo_state_test.c)
    xrdbg_enable(xrdbg_demo_state_test SAVE_STATES)
    set_property(TARGET xrdbg_demo_state_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    add_test(NAME xrdbg_demo_state COMMAND xrdbg_demo_state_test)
endif()

if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND BUILD_TESTING)
    enable_language(ASM_MASM)
    add_executable(xrdbg_native_test tests/native_test.c src/xrdbg_native.c src/xrdbg_stack_x64.asm)
    target_include_directories(xrdbg_native_test PRIVATE include)
    set_property(TARGET xrdbg_native_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_compile_options(xrdbg_native_test PRIVATE $<$<COMPILE_LANGUAGE:C>:/Zi>)
    target_link_options(xrdbg_native_test PRIVATE /DEBUG)
    add_test(NAME xrdbg_native COMMAND xrdbg_native_test)
    set_tests_properties(xrdbg_native PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
endif()
